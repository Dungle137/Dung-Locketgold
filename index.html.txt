      ownerLog.scrollTop = ownerLog.scrollHeight;

    function openGoldModal() {
      goldModal.classList.add("open");
      usernameInput.focus();
    }

      ownerModal.classList.add("open");
    }

    document.getElementById("btnOwnerPanel").addEventListener("click", openOwnerModal);

      m.addEventListener("click", (e) => {
        if (e.target === m) m.classList.remove("open");
      });
    });

      const user = usernameInput.value.trim() || "@dungle";
      const plan = document.getElementById("planSelect").value;

      goldModal.classList.remove("open");
